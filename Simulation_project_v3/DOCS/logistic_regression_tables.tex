\subsection*{回归方程}

\textbf{Logit形式：}
\begin{equation}
\begin{aligned}
\text{logit}(P(\text{matched}=1)) = \, &-5.735 + 0.129T + 0.040S + 0.128D \\
&- 0.001W - 0.162\sigma + 0.812\theta
\end{aligned}
\end{equation}

\textbf{概率形式：}
\begin{equation}
P(\text{matched}=1) = \frac{1}{1 + \exp(Z)}
\end{equation}

其中：
\begin{equation}
\begin{aligned}
Z = \, &5.735 - 0.129T - 0.040S - 0.128D \\
&+ 0.001W + 0.162\sigma - 0.812\theta
\end{aligned}
\end{equation}

\vspace{0.3cm}

变量说明：
\begin{itemize}
    \item $T$：劳动力工作时长（每周小时数）
    \item $S$：劳动力技能水平（1-10评分）
    \item $D$：劳动力数字素养（1-10评分）
    \item $W$：劳动力期望工资（元/小时）
    \item $\sigma$：控制变量综合指标（年龄、学历、孩子数量的二次标准化）
    \item $\theta$：市场紧张度（岗位数/求职者数）
\end{itemize}
